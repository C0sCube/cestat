CREATE TABLE `cog_dms_clb_nclt` (
   `Company_ID` int NOT NULL,
   `Board_ID` int NOT NULL,
   `Date` date NOT NULL,
   `Bench_ID` int DEFAULT NULL,
   `OrderType_ID` int DEFAULT NULL,
   `TypeofCase` varchar(250) NOT NULL DEFAULT '',
   `Remarks` varchar(5000) DEFAULT NULL,
   `Link` longtext,
   `Petitioner` varchar(250) DEFAULT NULL,
   `Respondent` varchar(250) DEFAULT NULL,
   `Amount` decimal(30,12) DEFAULT NULL,
   `Default_Keyperson_ID` int DEFAULT NULL,
   `Sources` varchar(250) DEFAULT NULL,
   `modified_date` timestamp NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
   PRIMARY KEY (`Company_ID`,`Board_ID`,`Date`,`TypeofCase`),
   KEY `fk_clbnclt_nw_Board_ID_idx` (`Board_ID`),
   KEY `fk_clbnclt_nw_Bench_ID_idx` (`Bench_ID`),
   KEY `fk_clbnclt_nw_OrderType_ID_idx` (`OrderType_ID`),
   KEY `clb_modified` (`modified_date`),
   KEY `fk_clbnclt_nw_OrderTypeID_idx` (`Company_ID`,`OrderType_ID`),
   CONSTRAINT `fk_clbnclt_nw_BenchID` FOREIGN KEY (`Bench_ID`) REFERENCES `cog_dms_master_clb_nclt_bench` (`ID`),
   CONSTRAINT `fk_clbnclt_nw_BoardID` FOREIGN KEY (`Board_ID`) REFERENCES `cog_dms_master_clb_nclt_board` (`ID`),
   CONSTRAINT `fk_clbnclt_nw_CompanyID` FOREIGN KEY (`Company_ID`) REFERENCES `dms_master_company` (`Company_ID`),
   CONSTRAINT `fk_clbnclt_nw_OrderTypeID` FOREIGN KEY (`OrderType_ID`) REFERENCES `cog_dms_master_clb_nclt_ordertype` (`id`)
 ) ENGINE=InnoDB DEFAULT CHARSET=latin1
